1. **开** ⇒ 报文含 `<current_date>YYYY-MM-DD</current_date>` 与 `<timezone>Asia/Shanghai</timezone>`；
2. **关** ⇒ 两块都**不出现**，且 `withoutDateAndTimezoneLines(enabled) == disabled` —— **其余模板逐字节相同**（关键回归断言；两次渲染共用同一个 CWD，避免临时目录差异造成的假失败）；

package appserver

import (
	"context"
	"os"
	"path/filepath"
	"strings"
	"testing"
	"time"

	"codex_go/compact"
	"codex_go/config"
	"codex_go/session"
	"codex_go/turn"

// environmentContextTimeClock is the fixed instant every case below renders.
var environmentContextTimeClock = time.Date(2026, 6, 17, 12, 0, 0, 0, time.UTC)

// environmentContextTimeRouter builds the per-turn router used by the
// include_environment_context_time cases.
func environmentContextTimeRouter(t *testing.T, cwd string) (*RuntimeRouter, *turn.TurnStartParams) {
	t.Helper()
	threadRouter := NewRouter(session.NewStore(filepath.Join(t.TempDir(), "sessions")))
	threadRouter.SetClock(func() time.Time { return environmentContextTimeClock })
	router := NewRuntimeRouter(RuntimeServices{ThreadRouter: threadRouter, DefaultCWD: cwd})
	return router, &turn.TurnStartParams{ThreadID: "thread-environment-time", CWD: cwd}
}

// turnEnvironmentContextTimeText renders one per-turn <environment_context>
// block for a config carrying include_environment_context_time. A nil value
// leaves the key absent, which is the unset-config default.
func turnEnvironmentContextTimeText(t *testing.T, cwd string, includeTime *bool) string {
	t.Helper()
	router, params := environmentContextTimeRouter(t, cwd)
	values := map[string]any{}
	if includeTime != nil {
		values["include_environment_context_time"] = *includeTime
	}
	items, _, err := router.turnEnvironmentContextInputItemForTurn(
		context.Background(), params.ThreadID, "turn-1", params, &config.Config{Values: values})
	if err != nil {
		t.Fatal(err)
	}
	texts := messageInputTextsForRole(items, "user")
	if len(texts) != 1 {
		t.Fatalf("environment context texts = %#v, want exactly one", texts)
	}
	return texts[0]
}

// withoutDateAndTimezoneLines drops the two gated lines so the remaining
// template can be compared byte for byte.
func withoutDateAndTimezoneLines(text string) string {
	lines := strings.Split(text, "\n")
	kept := make([]string, 0, len(lines))
	for _, line := range lines {
		if strings.Contains(line, "<current_date>") || strings.Contains(line, "<timezone>") {
			continue
		}
		kept = append(kept, line)
	}
	return strings.Join(kept, "\n")
}

// TestEnvironmentContextTimeEnabledWritesDateAndTimezoneLikeRust covers the
// flag-on arm of #52164: Rust keeps the date (core/src/session/world_state.rs:281)
// and the timezone (core/src/context/world_state/environment.rs:67).
func TestEnvironmentContextTimeEnabledWritesDateAndTimezoneLikeRust(t *testing.T) {
	enabled := true
	text := turnEnvironmentContextTimeText(t, t.TempDir(), &enabled)
	wantDate := environmentContextTimeClock.In(time.Local).Format("2006-01-02")
	if !strings.Contains(text, "<current_date>"+wantDate+"</current_date>") {
		t.Fatalf("environment context = %q, want <current_date>%s</current_date>", text, wantDate)
	}
	if !strings.Contains(text, "<timezone>"+localTimezoneName()+"</timezone>") {
		t.Fatalf("environment context = %q, want <timezone>%s</timezone>", text, localTimezoneName())
	}
}

// TestEnvironmentContextTimeDisabledOmitsDateAndTimezoneLikeRust covers the
// flag-off arm: Rust passes None for both values and push_optional_element
// (core/src/context/world_state/environment.rs:357-368) writes nothing, so the
// elements vanish instead of rendering empty. Everything else in the template
// must stay byte-identical - that is the regression this case guards.
func TestEnvironmentContextTimeDisabledOmitsDateAndTimezoneLikeRust(t *testing.T) {
	enabled, disabled := true, false
	cwd := t.TempDir()
	withTime := turnEnvironmentContextTimeText(t, cwd, &enabled)
	withoutTime := turnEnvironmentContextTimeText(t, cwd, &disabled)

	if strings.Contains(withoutTime, "<current_date>") || strings.Contains(withoutTime, "<timezone>") {
		t.Fatalf("disabled environment context = %q, want no <current_date>/<timezone>", withoutTime)
	}
	if want := withoutDateAndTimezoneLines(withTime); withoutTime != want {
		t.Fatalf("disabled environment context =\n%q\nwant the enabled template minus the two gated lines =\n%q", withoutTime, want)
	}
	if !strings.Contains(withoutTime, "<cwd>") || !strings.HasSuffix(withoutTime, "</environment_context>") {
		t.Fatalf("disabled environment context lost unrelated template content = %q", withoutTime)
	}
}

// TestEnvironmentContextTimeDefaultsTrueLikeRust covers the default arm: an
// absent key is true (core/src/config/mod.rs:4161 unwrap_or(true)), so the
// rendering must match the explicit-true rendering byte for byte.
func TestEnvironmentContextTimeDefaultsTrueLikeRust(t *testing.T) {
	enabled := true
	cwd := t.TempDir()
	explicit := turnEnvironmentContextTimeText(t, cwd, &enabled)
	unset := turnEnvironmentContextTimeText(t, cwd, nil)
	if unset != explicit {
		t.Fatalf("unset include_environment_context_time =\n%q\nwant the enabled rendering =\n%q", unset, explicit)
	}
	if !strings.Contains(unset, "<current_date>") || !strings.Contains(unset, "<timezone>") {
		t.Fatalf("unset environment context = %q, want date and timezone", unset)
	}
}

// compactEnvironmentContextTimeText renders the compaction-path consumer with a
// codex home whose config.toml carries include_environment_context_time. This
// exercises the second production call site, which resolves the flag through
// the turn's effective config snapshot instead of a hand-built config value.
func compactEnvironmentContextTimeText(t *testing.T, cwd string, includeTime string) string {

	}

	threadRouter.SetClock(func() time.Time { return environmentContextTimeClock })

		ThreadRouter: threadRouter,
		Config:       config.NewConfigService(home),

		t.Fatal(err)
	}
	if len(items) != 1 {
		t.Fatalf("compact environment context items = %#v, want exactly one", items)
	}
	return items[0].Text
}

}
